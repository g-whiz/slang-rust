cmake_minimum_required(VERSION 3.22)
project(slang_rhi_sys LANGUAGES C CXX)

set(SLANG_TREE_DIR "" CACHE PATH "The top level of the Slang source tree")
if(NOT IS_DIRECTORY "${SLANG_TREE_DIR}")
    message(FATAL_ERROR "SLANG_TREE_DIR is not a directory: ${SLANG_TREE_DIR}")
endif()
set(SLANG_INCLUDE_DIR "" CACHE PATH "The include directory of the Slang that the build links")
if(NOT IS_DIRECTORY "${SLANG_INCLUDE_DIR}")
    message(FATAL_ERROR "SLANG_INCLUDE_DIR is not a directory: ${SLANG_INCLUDE_DIR}")
endif()

add_library(slang INTERFACE)
target_include_directories(slang INTERFACE "${SLANG_INCLUDE_DIR}")
target_compile_definitions(slang INTERFACE SLANG_STATIC)

set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(BUILD_SHARED_LIBS OFF)

set(SLANG_RHI_BUILD_FROM_SLANG_REPO ON)
set(SLANG_RHI_BUILD_SHARED OFF)
set(SLANG_RHI_BUILD_TESTS OFF)
set(SLANG_RHI_BUILD_TESTS_WITH_GLFW OFF)
set(SLANG_RHI_BUILD_EXAMPLES OFF)
set(SLANG_RHI_INSTALL OFF)
set(SLANG_RHI_FETCH_SLANG OFF)
set(SLANG_RHI_FETCH_DXC OFF)
set(SLANG_RHI_SLANG_INCLUDE_DIR "${SLANG_INCLUDE_DIR}")

set(SLANG_RHI_ENABLE_VULKAN ON)
set(SLANG_RHI_ENABLE_CPU OFF)
set(SLANG_RHI_ENABLE_D3D11 OFF)
set(SLANG_RHI_ENABLE_D3D12 OFF)
set(SLANG_RHI_ENABLE_AGILITY_SDK OFF)
set(SLANG_RHI_ENABLE_NVAPI OFF)
set(SLANG_RHI_ENABLE_METAL OFF)
set(SLANG_RHI_ENABLE_CUDA OFF)
set(SLANG_RHI_ENABLE_OPTIX OFF)
set(SLANG_RHI_ENABLE_WGPU OFF)
set(SLANG_RHI_ENABLE_AFTERMATH OFF)

include(CheckIncludeFileCXX)
check_include_file_cxx(X11/Xlib.h SLANG_RHI_SYS_HAVE_XLIB_H)
if(NOT SLANG_RHI_SYS_HAVE_XLIB_H)
    message(FATAL_ERROR "The C++ compiler cannot find the header X11/Xlib.h. The Vulkan backend of slang-rhi includes X11/Xlib.h on Linux, so the build of slang-rhi needs the development headers of Xlib, such as those of the Debian package libx11-dev. Install them and build again.")
endif()

add_subdirectory("${SLANG_TREE_DIR}/external/slang-rhi" slang-rhi EXCLUDE_FROM_ALL)

set_property(TARGET slang-rhi-warnings-as-errors PROPERTY INTERFACE_COMPILE_OPTIONS "")

set(SLANG_RHI_SYS_ABI_PROBE "${CMAKE_BINARY_DIR}/abi-probe.cpp")
set_source_files_properties("${SLANG_RHI_SYS_ABI_PROBE}" PROPERTIES GENERATED TRUE)
add_library(slang-rhi-sys-abi-probe STATIC EXCLUDE_FROM_ALL "${SLANG_RHI_SYS_ABI_PROBE}")
set_target_properties(slang-rhi-sys-abi-probe PROPERTIES CXX_STANDARD 20 CXX_STANDARD_REQUIRED ON)
target_link_libraries(slang-rhi-sys-abi-probe PRIVATE slang-rhi)
